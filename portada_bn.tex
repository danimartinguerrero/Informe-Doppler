      \pgfmathsetmacro{\lw}{0.25+0.04*\k}

        ($(O)+(\cx cm,17.2cm)$) circle (\rr cm);

      ($(O)+(13.24cm,17.2cm)$) -- ($(O)+(14.9cm,17.2cm)$);

    \node[azulprof, anchor=south, inner sep=2pt, font=\itshape\normalsize]
