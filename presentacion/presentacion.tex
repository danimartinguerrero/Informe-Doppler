\section{Objetivo}
% =========================================================================
\begin{frame}{Objetivo}
  Cuando hacemos una gráfica con ajuste, el código se encarga de dos tareas:
  \begin{block}{}
    \begin{itemize}
      \item \resalta{Calcular la recta} que mejor describe los datos, teniendo en
        cuenta lo fiable que es cada punto.
      \item \resalta{Presentar el resultado} con el número correcto de cifras
        significativas.
    \end{itemize}
  \end{block}
  Vamos a recorrer el código línea a línea con los números reales del ajuste de
  la velocidad del sonido (caso~1 del informe): $v$ frente a $x=1-\fE/\fR$.
\end{frame}

\section{El redondeo}
% =========================================================================
\begin{frame}{Criterio de redondeo}
  \begin{block}{}
    \resalta{Criterio del informe:}
    \begin{itemize}
      \item la incertidumbre se da con \emph{una} cifra significativa;
      \item si esa cifra es un 1, se conservan \emph{dos};
      \item el valor se redondea a la misma posición decimal.
    \end{itemize}
  \end{block}
  \centering\small
  \begin{tabular}{@{}lcl@{}}
    \toprule
    Sin redondear & 1.ª cifra & Resultado\\
    \midrule
    A: $0{,}353\,17\pm0{,}011\,8$ & 1 & $0{,}353\pm0{,}012$\\
    B: $244{,}66\pm88{,}6$ & 8 & $(2{,}4\pm0{,}9)\times10^{2}$\\
    \bottomrule
  \end{tabular}
  \par\vspace{4pt}
  \raggedright\small Los ejemplos A y B nos acompañarán en toda esta sección.
\end{frame}

\begin{frame}[fragile]{La función \texttt{redondea}}
  \begin{codigo}[listing options={style=python, basicstyle=\ttfamily\scriptsize\color{codtexto}}]
def redondea(valor, incertidumbre):
    exp = math.floor(math.log10(abs(incertidumbre)))
    primera = int(abs(incertidumbre) / 10**exp)
    decimales = -(exp - 1) if primera == 1 else -exp
    v, u = round(valor, decimales), round(incertidumbre, decimales)
    d = max(decimales, 0)
    return f"{v:.{d}f}".replace(".", ","), f"{u:.{d}f}".replace(".", ",")
  \end{codigo}
  Recibe el valor y su incertidumbre \resalta{sin redondear} y devuelve dos
  textos listos para el informe. Los pasos son:
  \begin{enumerate}
    \item localizar la primera cifra significativa;
    \item ver si es un 1;
    \item decidir dónde cortar;
    \item redondear y escribir con coma decimal.
  \end{enumerate}
\end{frame}

\begin{frame}[fragile]{Paso 1: posición de la primera cifra}
  \begin{codigo}
exp = math.floor(math.log10(abs(incertidumbre)))
  \end{codigo}
  El \resalta{logaritmo decimal} responde a la pregunta: \emph{¿a qué número hay
  que elevar 10 para obtener este?}
  \begin{columns}[T]
    \begin{column}{0.4\textwidth}
      \centering\small
      \begin{tabular}{@{}rcc@{}}
        \toprule
        Número & Potencia & $\log_{10}$\\
        \midrule
        100 & $10^{2}$ & 2\\
        10 & $10^{1}$ & 1\\
        1 & $10^{0}$ & 0\\
        0,1 & $10^{-1}$ & $-1$\\
        0,01 & $10^{-2}$ & $-2$\\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.56\textwidth}
      \small Si el número cae entre dos potencias, su logaritmo cae entre los dos
      enteros:
      \begin{itemize}
        \item $\log_{10}(88{,}6)\approx 1{,}95$;
        \item $\log_{10}(0{,}0118)\approx -1{,}93$.
      \end{itemize}
      \cod{abs(...)} garantiza que el número sea positivo.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Paso 1: el papel de \texttt{floor}}
  \cod{math.floor} redondea \resalta{siempre hacia abajo} y conserva la parte
  entera, que indica la posición de la primera cifra significativa:
  \begin{block}{}
    \centering
    \begin{tabular}{@{}lccl@{}}
      & $\log_{10}$ & \texttt{floor} & Posición\\
      \midrule
      A: $0{,}0118$ & $-1{,}93$ & $-2$ & centésimas ($10^{-2}$)\\
      B: $88{,}6$ & $1{,}95$ & $1$ & decenas ($10^{1}$)\\
    \end{tabular}
  \end{block}
  \begin{itemize}
    \item En los negativos, «hacia abajo» significa \emph{alejándose del cero}:
      $-1{,}93$ pasa a $-2$, no a $-1$, y así debe ser, porque el 1 de $0{,}0118$
      está en las centésimas.
    \item A ojo es fácil ver dónde empieza un número; el logaritmo es la forma
      matemática de que el ordenador lo averigüe para cualquier valor.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Paso 2: valor de la primera cifra}
  \begin{codigo}
primera = int(abs(incertidumbre) / 10**exp)
  \end{codigo}
  Dividir entre $10^{\text{exp}}$ desplaza la coma hasta dejar la primera cifra
  delante, e \cod{int(...)} corta todo lo que hay detrás:
  \begin{block}{}
    \begin{equation}
      \text{A: }\frac{0{,}0118}{10^{-2}}=1{,}18\;\rightarrow\;1,
      \qquad
      \text{B: }\frac{88{,}6}{10^{1}}=8{,}86\;\rightarrow\;8.
    \end{equation}
  \end{block}
  Ya sabemos \resalta{dónde está} la primera cifra (\texttt{exp}) y
  \resalta{cuál es} (\texttt{primera}): con eso basta para aplicar el criterio.
\end{frame}

\begin{frame}[fragile]{Paso 3: posición de corte}
  \begin{codigo}
decimales = -(exp - 1) if primera == 1 else -exp
  \end{codigo}
  \begin{itemize}
    \item Si la primera cifra es un 1, se conservan \resalta{dos} cifras:
      se corta una posición más a la derecha, \cod{-(exp - 1)}.
    \item En otro caso se conserva \resalta{una}: \cod{-exp}.
  \end{itemize}
  \begin{block}{}
    \centering
    \begin{tabular}{@{}lccc@{}}
      & \texttt{exp} & \texttt{primera} & \texttt{decimales}\\
      \midrule
      A & $-2$ & 1 & $-(-2-1)=3$ \quad (milésimas)\\
      B & $1$ & 8 & $-1$ \quad (decenas)\\
    \end{tabular}
  \end{block}
  Un número de decimales negativo indica que se redondea a la izquierda de la
  coma.
\end{frame}

\begin{frame}[fragile]{Paso 4: redondeo y escritura}
  \begin{codigo}[listing options={style=python, basicstyle=\ttfamily\scriptsize\color{codtexto}}]
v, u = round(valor, decimales), round(incertidumbre, decimales)
d = max(decimales, 0)
return f"{v:.{d}f}".replace(".", ","), f"{u:.{d}f}".replace(".", ",")
  \end{codigo}
  \begin{itemize}
    \item \cod{round} redondea valor e incertidumbre a la \resalta{misma
      posición}.
    \item \cod{max(decimales, 0)}: si se redondea a las decenas, no hay
      decimales que escribir.
    \item El formato escribe exactamente \texttt{d} decimales y \cod{replace}
      cambia el punto por la coma.
  \end{itemize}
  \begin{block}{}
    \centering
    \begin{tabular}{@{}lll@{}}
      & Entrada & Devuelve\\
      \midrule
      A & $0{,}353\,17$; $0{,}011\,8$ & \texttt{0,353} y \texttt{0,012}\\
      B & $244{,}66$; $88{,}6$ & \texttt{240} y \texttt{90}, es decir,
      $(2{,}4\pm0{,}9)\times10^{2}$\\
    \end{tabular}
  \end{block}
\end{frame}

\begin{frame}[fragile]{La función \texttt{coma} y dos detalles}
  \begin{codigo}
def coma(x, d):
    return f"{x:.{d}f}".replace(".", ",")
  \end{codigo}
  Solo da formato, sin aplicar el criterio de redondeo; se usa para números
  auxiliares como $t_{95,\nu}$ o $\chi^{2}/\nu$.
  \begin{block}{}
    \resalta{Empates:} Python lleva los 5 exactos a la cifra par
    (\cod{round(0.125, 2)} da $0{,}12$), como indica la regla de las
    diapositivas de teoría de errores.
  \end{block}
  \begin{block}{}
    \resalta{Caso límite:} $u=0{,}0196$ empieza por 1 y conserva dos cifras,
    $0{,}020$; tras redondear empieza por 2, pero mantiene las dos. Es poco
    frecuente y, si aparece, se corrige a mano.
  \end{block}
\end{frame}

\section{El ajuste ponderado}
% =========================================================================
\begin{frame}{Motivación}
  Cada punto es la media de seis lecturas, con $u^{2}=s^{2}/n+\Delta^{2}/12$, y la
  abscisa $x=1-\fE/\fR$ hereda la incertidumbre de las frecuencias:
  \begin{center}\small
  \begin{tabular}{@{}lcc@{}}
    \toprule
    Serie & $u(y)$ (m/s) & $a\,u(x)$ (m/s)\\
    \midrule
    Máxima, acercamiento & 0,0026 & 0,030\\
    Máxima, alejamiento & 0,0012 & \resalta{0,184}\\
    Media, acercamiento & 0,0009 & 0,011\\
    Media, alejamiento & 0,0008 & 0,015\\
    Mínima, acercamiento & 0,0004 & 0,025\\
    Mínima, alejamiento & 0,0007 & 0,008\\
    \bottomrule
  \end{tabular}
  \end{center}
  \begin{block}{}
    La incertidumbre horizontal, trasladada a la vertical, es entre 10 y 150 veces
    mayor: el ajuste no puede ignorar la $x$ ni tratar igual todos los puntos.
  \end{block}
\end{frame}

\begin{frame}[fragile]{La función \texttt{ajuste\_ponderado} (I)}
  \begin{codigo}[listing options={style=python, basicstyle=\ttfamily\scriptsize\color{codtexto}}]
def ajuste_ponderado(x, ux, y, uy, modelo="libre"):
    x, ux, y, uy = (np.asarray(t, float) for t in (x, ux, y, uy))
    n = len(x)
    a = np.polyfit(x, y, 1)[0] if modelo == "libre" else (x @ y) / (x @ x)
    for _ in range(500):
        w = 1 / (uy**2 + a**2 * ux**2)
        if modelo == "libre":
            W = w.sum()
            m = lambda z: (w * z).sum() / W
            a_nueva = (m(x * y) - m(x) * m(y)) / (m(x * x) - m(x) ** 2)
        else:
            a_nueva = (w * x * y).sum() / (w * x * x).sum()
        if abs(a_nueva - a) <= 1e-12 * max(1.0, abs(a)):
            a = a_nueva
            break
        a = a_nueva
  \end{codigo}
\end{frame}

\begin{frame}[fragile]{La función \texttt{ajuste\_ponderado} (II)}
  \begin{codigo}[listing options={style=python, basicstyle=\ttfamily\scriptsize\color{codtexto}}]
ue = np.sqrt(uy**2 + a**2 * ux**2)
w = 1 / ue**2
if modelo == "libre":
    W = w.sum()
    m = lambda z: (w * z).sum() / W
    b = m(y) - a * m(x)
    ua = 1 / math.sqrt(W) / math.sqrt(m(x * x) - m(x) ** 2)
    ub = ua * math.sqrt(m(x * x))
    nu = n - 2
else:
    b, ub, nu = 0.0, 0.0, n - 1
    ua = 1 / math.sqrt((w * x * x).sum())
chi2 = float((w * (y - a * x - b) ** 2).sum())
factor = max(1.0, math.sqrt(chi2 / nu)) if nu > 0 else 1.0
t95 = float(stats.t.ppf(0.975, nu)) if nu > 0 else float("nan")
  \end{codigo}
\end{frame}

\begin{frame}[fragile]{Preparación y pendiente inicial}
  \begin{codigo}[listing options={style=python, basicstyle=\ttfamily\scriptsize\color{codtexto}}]
x, ux, y, uy = (np.asarray(t, float) for t in (x, ux, y, uy))
n = len(x)
a = np.polyfit(x, y, 1)[0] if modelo == "libre" else (x @ y) / (x @ x)
  \end{codigo}
  \begin{itemize}
    \item \cod{np.asarray} convierte las listas en \emph{arrays}, con los que se
      opera punto a punto: \cod{x * y} multiplica cada $x_i$ por su $y_i$.
    \item \cod{n = len(x)} cuenta los puntos: $n=6$.
    \item La iteración necesita una pendiente de partida:
      \begin{itemize}
        \item modelo libre: el ajuste sin pesos de \cod{np.polyfit};
        \item recta por el origen: $a=\sum x_iy_i\big/\sum x_i^{2}$.
      \end{itemize}
  \end{itemize}
  \begin{block}{}
    En el caso~1 la pendiente de partida es $a_{0}=82{,}4$~m/s: el ajuste básico,
    arrastrado por el punto anómalo.
  \end{block}
\end{frame}

\begin{frame}[fragile]{Pesos y varianza efectiva}
  \begin{codigo}
w = 1 / (uy**2 + a**2 * ux**2)
  \end{codigo}
  \begin{block}{}
    \resalta{Peso de cada punto:}
    \begin{equation}
      w_i=\frac{1}{u_i^{2}},\qquad u_i^{2}=u^{2}(y_i)+a^{2}\,u^{2}(x_i).
    \end{equation}
  \end{block}
  \begin{itemize}
    \item Un punto con el doble de incertidumbre pesa \resalta{cuatro veces
      menos}: los puntos precisos mandan en la recta.
    \item ¿Por qué $a^{2}$? Un desplazamiento horizontal $\delta x$ equivale, sobre
      una recta de pendiente $a$, a uno vertical $a\,\delta x$; sumando ambas
      contribuciones como varianzas independientes se obtiene $u_i^{2}$.
    \item Para calcular $w_i$ hace falta $a$, que es justo lo que se busca: por eso
      hay que \resalta{iterar}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{La media ponderada}
  \begin{codigo}
W = w.sum()
m = lambda z: (w * z).sum() / W
  \end{codigo}
  \cod{lambda} define una función en una línea; \texttt{z} es solo el nombre de
  su argumento. \cod{m} calcula la \resalta{media ponderada}:
  \begin{block}{}
    \begin{equation}
      \langle z\rangle=\frac{\sum_i w_i\,z_i}{\sum_i w_i}.
    \end{equation}
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \centering\small
      \begin{tabular}{@{}llr@{}}
        \toprule
        Llamada & Calcula & Caso 1\\
        \midrule
        \cod{m(x)} & $\langle x\rangle$ & $3{,}27\times10^{-4}$\\
        \cod{m(y)} & $\langle y\rangle$ & $-0{,}0214$\\
        \cod{m(x*x)} & $\langle x^{2}\rangle$ & $6{,}49\times10^{-7}$\\
        \cod{m(x*y)} & $\langle xy\rangle$ & $1{,}26\times10^{-4}$\\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.46\textwidth}
      \small Ejemplo: con pesos $(1,1,2)$ y valores $(10,20,40)$,
      $\langle z\rangle=110/4=27{,}5$, mayor que la media aritmética
      ($23{,}3$), porque el 40 pesa el doble.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{La pendiente ponderada}
  \begin{codigo}[listing options={style=python, basicstyle=\ttfamily\scriptsize\color{codtexto}}]
a_nueva = (m(x * y) - m(x) * m(y)) / (m(x * x) - m(x) ** 2)
  \end{codigo}
  \begin{block}{}
    \resalta{Mínimos cuadrados ponderados:} minimizando
    $\chi^{2}=\sum_i w_i\,(y_i-ax_i-b)^{2}$ respecto de $a$ y $b$,
    \begin{equation}
      a=\frac{\langle xy\rangle-\langle x\rangle\langle y\rangle}
             {\langle x^{2}\rangle-\langle x\rangle^{2}},
      \qquad
      b=\langle y\rangle-a\langle x\rangle .
    \end{equation}
  \end{block}
  \begin{itemize}
    \item El numerador es una «covarianza» ponderada entre $x$ e $y$, y el
      denominador, una «varianza» ponderada de $x$.
    \item Con todos los pesos iguales se recuperan las fórmulas básicas de las
      notas de incertidumbres.
    \item Para la recta por el origen: \cod{a\_nueva = (w*x*y).sum() / (w*x*x).sum()}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Criterio de convergencia}
  \begin{codigo}
if abs(a_nueva - a) <= 1e-12 * max(1.0, abs(a)):
    a = a_nueva
    break
a = a_nueva
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small Si la pendiente nueva apenas difiere de la anterior (una billonésima
      relativa), el cálculo ha convergido y se sale con \cod{break}; si no, se
      guarda y se da otra vuelta. \cod{range(500)} es solo un tope de
      seguridad.
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering\small
      \begin{tabular}{@{}lc@{}}
        \toprule
        Vuelta & $a$ (m/s)\\
        \midrule
        Inicio & 82,4\\
        1 & 244,279\\
        2 & 244,657\\
        3 & 244,657\\
        \bottomrule
      \end{tabular}
    \end{column}
  \end{columns}
  \begin{block}{}

\begin{frame}[fragile]{Ordenada e incertidumbres de los parámetros}

\begin{frame}[fragile]{Bondad del ajuste: $\chi^{2}$}
  \begin{codigo}[listing options={style=python, basicstyle=\ttfamily\scriptsize\color{codtexto}}]
chi2 = float((w * (y - a * x - b) ** 2).sum())
factor = max(1.0, math.sqrt(chi2 / nu)) if nu > 0 else 1.0
  \end{codigo}
  \begin{block}{}
    \begin{equation}
      \chi^{2}=\sum_i\left(\frac{y_i-ax_i-b}{u_i}\right)^{2}.
    \end{equation}
  \end{block}
  \small
  \begin{itemize}
    \item Si las barras de error están bien estimadas, $\chi^{2}\approx\nu$.
    \item En el caso~1, $\chi^{2}=69{,}3$ frente a $\nu=4$: la dispersión supera lo
      que indican las barras y $u(a)=7{,}67$~m/s es demasiado optimista.
    \item Se corrige con $\sqrt{69{,}3/4}=4{,}16$; \cod{max(1.0, ...)} impide que la
      incertidumbre disminuya.
  \end{itemize}
\end{frame}
