\section{El redondeo}
% =========================================================================

\begin{frame}{La regla que seguimos}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{formula}[Criterio del informe]
        \begin{itemize}
          \item La incertidumbre se da con \textbf{una} cifra significativa.
          \item Si esa cifra es un \textbf{1}, conservamos \textbf{dos}.
          \item El valor se redondea a la \textbf{misma posición decimal} que la
            incertidumbre.
        \end{itemize}
      \end{formula}
      \small Con una sola cifra, $0{,}0118$ quedaría en $0{,}01$ y perderíamos
      casi un 20\,\% de información; por eso, si empieza por 1, guardamos dos.
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering\footnotesize
      \setlength{\tabcolsep}{4pt}
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{@{}lcl@{}}
        \toprule
        Sin redondear & 1.ª cifra & Resultado\\
        \midrule
        $0{,}353\,17 \pm 0{,}011\,8$ & 1 & $0{,}353 \pm 0{,}012$\\
        $244{,}66 \pm 88{,}6$ & 8 & $(2{,}4 \pm 0{,}9)\times10^{2}$\\
        $40\,607{,}67 \pm 0{,}357$ & 3 & $40\,607{,}7 \pm 0{,}4$\\
        \bottomrule
      \end{tabular}
      \par\vspace{8pt}
      \begin{destacado}
        \small Los dos primeros ejemplos, A ($u=0{,}0118$) y B ($u=88{,}6$),
        nos acompañarán durante toda esta parte.
      \end{destacado}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{La función completa}
  \begin{codigo}[listing options={style=python, basicstyle=\ttfamily\footnotesize\color{codtexto}}]
def redondea(valor, incertidumbre):
    exp = math.floor(math.log10(abs(incertidumbre)))
    primera = int(abs(incertidumbre) / 10**exp)
    decimales = -(exp - 1) if primera == 1 else -exp
    v, u = round(valor, decimales), round(incertidumbre, decimales)
    d = max(decimales, 0)
    return f"{v:.{d}f}".replace(".", ","), f"{u:.{d}f}".replace(".", ",")
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small Recibe el valor y su incertidumbre \textbf{sin redondear} y
      devuelve dos textos listos para escribir en el informe.
    \end{column}
    \begin{column}{0.48\textwidth}
      \small Los pasos son: localizar la primera cifra, ver si es un 1,
      decidir dónde cortar, redondear y escribir con coma decimal.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Paso 1: ¿dónde está la primera cifra?}
  \begin{codigo}
exp = math.floor(math.log10(abs(incertidumbre)))
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small El \textbf{logaritmo decimal} responde a una pregunta muy concreta:
      \emph{¿a qué número tengo que elevar 10 para obtener este?}
      \par\vspace{6pt}\centering
      \renewcommand{\arraystretch}{1.15}
      \begin{tabular}{@{}rcc@{}}
        \toprule
        Número & Potencia & $\log_{10}$\\
        \midrule
        100 & $10^{2}$ & 2\\
        10 & $10^{1}$ & 1\\
        1 & $10^{0}$ & 0\\
        0,1 & $10^{-1}$ & $-1$\\
        0,01 & $10^{-2}$ & $-2$\\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.46\textwidth}
      \small Si el número cae entre dos potencias, su logaritmo cae entre los
      dos enteros:
      \begin{itemize}
        \item $88{,}6$ está entre 10 y 100: $\log_{10}(88{,}6)\approx 1{,}95$.
        \item $0{,}0118$ está entre $0{,}01$ y $0{,}1$:
          $\log_{10}(0{,}0118)\approx -1{,}93$.
      \end{itemize}
      \cod{abs(...)} solo garantiza que el número sea positivo, porque el
      logaritmo de un negativo no existe.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Paso 1: el papel de \texttt{floor}}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small \cod{math.floor} redondea \textbf{siempre hacia abajo} y se queda
      con la parte entera, que indica la \textbf{posición de la primera cifra
      significativa}:
      \par\vspace{6pt}\centering
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{@{}lccl@{}}
        \toprule
        & $\log_{10}$ & \texttt{floor} & Posición\\
        \midrule
        A: $0{,}0118$ & $-1{,}93$ & $-2$ & centésimas ($10^{-2}$)\\
        B: $88{,}6$ & $1{,}95$ & $1$ & decenas ($10^{1}$)\\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{destacado}[Ojo con los negativos]
        \small «Hacia abajo» significa \emph{alejándose del cero}: $-1{,}93$ pasa
        a $-2$, no a $-1$. Y así es como debe ser, porque el 1 de $0{,}0118$
        está en las centésimas, no en las décimas.
      \end{destacado}
      \small A ojo es fácil ver dónde empieza un número; el ordenador no lo
      «ve», y el logaritmo es la manera matemática de averiguarlo para
      cualquier valor.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Paso 2: ¿cuál es esa cifra?}
  \begin{codigo}
primera = int(abs(incertidumbre) / 10**exp)
  \end{codigo}
  \small Dividir entre $10^{\text{exp}}$ «corre la coma» hasta dejar la primera
  cifra delante, e \cod{int(...)} corta todo lo que hay detrás:
  \begin{equation*}
    \text{A: } \frac{0{,}0118}{10^{-2}} = 1{,}18 \;\longrightarrow\; \mathbf{1},
    \qquad\qquad
    \text{B: } \frac{88{,}6}{10^{1}} = 8{,}86 \;\longrightarrow\; \mathbf{8}.
  \end{equation*}
  \begin{destacado}
    \small Ya sabemos dónde está la primera cifra (\texttt{exp}) y cuál es
    (\texttt{primera}). Con eso basta para aplicar la regla.
  \end{destacado}
\end{frame}

\begin{frame}[fragile]{Paso 3: ¿dónde cortamos?}
  \begin{codigo}
decimales = -(exp - 1) if primera == 1 else -exp
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small Aquí vive la regla:
      \begin{itemize}
        \item Si la primera cifra es un 1, guardamos \textbf{dos} cifras:
          cortamos una posición más a la derecha, \cod{-(exp - 1)}.
        \item Si no, guardamos \textbf{una}: \cod{-exp}.
      \end{itemize}
      \small Un número de decimales negativo significa redondear a la izquierda
      de la coma: $-1$ es «a las decenas».
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering\small
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{@{}lccc@{}}
        \toprule
        & \texttt{exp} & \texttt{primera} & \texttt{decimales}\\
        \midrule
        A & $-2$ & 1 & $-(-2-1)=3$\\
        B & $1$ & 8 & $-1$\\
        \bottomrule
      \end{tabular}
      \par\vspace{8pt}
      \begin{destacado}
        \small A: tres decimales, hasta las milésimas.\\
        B: hasta las decenas.
      \end{destacado}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Paso 4: redondear y escribir}
  \begin{codigo}
v, u = round(valor, decimales), round(incertidumbre, decimales)
d = max(decimales, 0)
return f"{v:.{d}f}".replace(".", ","), f"{u:.{d}f}".replace(".", ",")
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      \begin{itemize}
        \item \cod{round} redondea valor e incertidumbre \textbf{a la misma
          posición}, para que sus últimas cifras coincidan.
        \item \cod{max(decimales, 0)}: si redondeamos a las decenas no hay
          decimales que escribir, así que usamos 0.
        \item \cod{f\textquotedbl\{v:.\{d\}f\}\textquotedbl} escribe exactamente \texttt{d} decimales
          (conserva ceros como el de $0{,}0120$) y \cod{replace} pone la coma.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering\small
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{@{}lll@{}}
        \toprule
        & Entrada & Devuelve\\
        \midrule
        A & $0{,}353\,17$, $0{,}011\,8$ & \texttt{"0,353"}, \texttt{"0,012"}\\
        B & $244{,}66$, $88{,}6$ & \texttt{"240"}, \texttt{"90"}\\
        \bottomrule
      \end{tabular}
      \par\vspace{6pt}
      \small El caso B lo escribimos en el informe como
      $(2{,}4\pm0{,}9)\times10^{2}$\,m/s.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{La función \texttt{coma} y dos detalles finos}
  \begin{codigo}
def coma(x, d):
    return f"{x:.{d}f}".replace(".", ",")
  \end{codigo}
  \small Solo da formato: escribe \texttt{x} con \texttt{d} decimales y coma
  decimal, sin aplicar ninguna regla. La uso para números auxiliares como
  $t_{95,\nu}$ o $\chi^{2}/\nu$: \cod{coma(2.5706, 2)} devuelve \texttt{"2,57"}.
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{formula}[Redondeo de los empates]
        \small Python lleva los 5 exactos a la cifra \emph{par}:
        \cod{round(0.125, 2)} da $0{,}12$. Coincide con la regla de las
        diapositivas («si es 5, según sea par o impar»).
      \end{formula}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{formula}[Un caso límite]
        \small $u=0{,}0196$ empieza por 1 y conserva dos cifras: $0{,}020$. Tras
        redondear empieza por 2, pero mantiene las dos. Es raro; si aparece,
        se ajusta a mano.
      \end{formula}
    \end{column}
  \end{columns}
\end{frame}

\section{El ajuste por mínimos cuadrados ponderados}
% =========================================================================

\begin{frame}{¿Por qué ponderar?}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \small En el caso 1 ajustamos $v$ frente a $x=1-\fE/\fR$. Cada punto es la
      media de seis lecturas, con $u^{2}=s^{2}/n+\Delta^{2}/12$, y la $x$ hereda
      la incertidumbre de las frecuencias. Comparemos las dos:
      \begin{destacado}
        \small La incertidumbre horizontal, llevada a la vertical, es
        \textbf{entre 10 y 150 veces mayor} que la vertical. El ajuste no puede
        ignorar la $x$, ni tratar igual todos los puntos.
      \end{destacado}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering\small
      \renewcommand{\arraystretch}{1.18}
      \begin{tabular}{@{}lcc@{}}
        \toprule
        Serie & $u(y)$ (m/s) & $a\,u(x)$ (m/s)\\
        \midrule
        Máxima, acercamiento & 0,0026 & 0,030\\
        Máxima, alejamiento & 0,0012 & \textcolor{acento}{\textbf{0,184}}\\
        Media, acercamiento & 0,0009 & 0,011\\
        Media, alejamiento & 0,0008 & 0,015\\
        Mínima, acercamiento & 0,0004 & 0,025\\
        Mínima, alejamiento & 0,0007 & 0,008\\
        \bottomrule
      \end{tabular}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{La función completa (I): la iteración}
  \begin{codigo}[listing options={style=python, basicstyle=\ttfamily\scriptsize\color{codtexto}}]
def ajuste_ponderado(x, ux, y, uy, modelo="libre"):
    x, ux, y, uy = (np.asarray(t, float) for t in (x, ux, y, uy))
    n = len(x)
    a = np.polyfit(x, y, 1)[0] if modelo == "libre" else (x @ y) / (x @ x)
    for _ in range(500):
        w = 1 / (uy**2 + a**2 * ux**2)
        if modelo == "libre":
            W = w.sum()
            m = lambda z: (w * z).sum() / W
            a_nueva = (m(x * y) - m(x) * m(y)) / (m(x * x) - m(x) ** 2)
        else:
            a_nueva = (w * x * y).sum() / (w * x * x).sum()
        if abs(a_nueva - a) <= 1e-12 * max(1.0, abs(a)):
            a = a_nueva
            break
        a = a_nueva
  \end{codigo}
\end{frame}

\begin{frame}[fragile]{La función completa (II): resultados}
  \begin{codigo}[listing options={style=python, basicstyle=\ttfamily\scriptsize\color{codtexto}}]
ue = np.sqrt(uy**2 + a**2 * ux**2)
w = 1 / ue**2
if modelo == "libre":
    W = w.sum()
    m = lambda z: (w * z).sum() / W
    b = m(y) - a * m(x)
    ua = 1 / math.sqrt(W) / math.sqrt(m(x * x) - m(x) ** 2)
    ub = ua * math.sqrt(m(x * x))
    nu = n - 2
else:
    b, ub, nu = 0.0, 0.0, n - 1
    ua = 1 / math.sqrt((w * x * x).sum())
chi2 = float((w * (y - a * x - b) ** 2).sum())
factor = max(1.0, math.sqrt(chi2 / nu)) if nu > 0 else 1.0
t95 = float(stats.t.ppf(0.975, nu)) if nu > 0 else float("nan")
  \end{codigo}
\end{frame}

\begin{frame}[fragile]{Preparación y pendiente de partida}
  \begin{codigo}
x, ux, y, uy = (np.asarray(t, float) for t in (x, ux, y, uy))
n = len(x)
a = np.polyfit(x, y, 1)[0] if modelo == "libre" else (x @ y) / (x @ x)
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      \begin{itemize}
        \item \cod{np.asarray} convierte las listas en \emph{arrays}, con los
          que se opera punto a punto: \cod{x * y} multiplica cada $x_i$ por su
          $y_i$.
        \item \cod{n = len(x)} cuenta los puntos: $n=6$.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \small La iteración necesita una pendiente inicial:
      \begin{itemize}
        \item \textbf{libre}: el ajuste sin pesos, \cod{np.polyfit}, que da
          $a_{0}=82{,}4$\,m/s;
        \item \textbf{origen}: $a=\sum x_iy_i\big/\sum x_i^{2}$.
      \end{itemize}
      \begin{destacado}
        \small Ese $82{,}4$\,m/s es el ajuste básico del informe, arrastrado
        por el punto anómalo.
      \end{destacado}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Los pesos: la varianza efectiva}
  \begin{codigo}
w = 1 / (uy**2 + a**2 * ux**2)
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{formula}[Peso de cada punto]
        \begin{equation*}
          w_i=\frac{1}{u_i^{2}},\qquad u_i^{2}=u^{2}(y_i)+a^{2}u^{2}(x_i)
        \end{equation*}
      \end{formula}
      \small Un punto con el doble de incertidumbre pesa \textbf{cuatro veces
      menos}: los precisos mandan en la recta.
    \end{column}
    \begin{column}{0.46\textwidth}
      \small ¿Por qué $a^{2}$? Si muevo un punto $\delta x$ en horizontal, sobre
      una recta de pendiente $a$ eso equivale a moverlo $a\,\delta x$ en
      vertical. Sumando ambas contribuciones como varianzas independientes
      sale la expresión de la izquierda.
      \begin{destacado}
        \small El problema: para calcular $w_i$ necesito $a$, que es justo lo que
        busco. Por eso hay que iterar.
      \end{destacado}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{La media ponderada: ¿qué es \texttt{z}?}
  \begin{codigo}
W = w.sum()
m = lambda z: (w * z).sum() / W
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small \cod{lambda} define una función en una línea; \texttt{z} es solo el
      nombre de su argumento, un \textbf{comodín}. \cod{m} es una máquina de
      medias ponderadas:
      \begin{equation*}
        \langle z\rangle=\frac{\sum_i w_iz_i}{\sum_i w_i}
      \end{equation*}
      \small Ejemplo: pesos $(1,1,2)$ y valores $(10,20,40)$ dan
      $\langle z\rangle=110/4=27{,}5$, más que la media normal ($23{,}3$),
      porque el 40 pesa el doble.
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering\small
      \renewcommand{\arraystretch}{1.25}
      \begin{tabular}{@{}llr@{}}
        \toprule
        Llamada & Calcula & Caso 1\\
        \midrule
        \cod{m(x)} & $\langle x\rangle$ & $3{,}27\times10^{-4}$\\
        \cod{m(y)} & $\langle y\rangle$ & $-0{,}0214$\,m/s\\
        \cod{m(x * x)} & $\langle x^{2}\rangle$ & $6{,}49\times10^{-7}$\\
        \cod{m(x * y)} & $\langle xy\rangle$ & $1{,}26\times10^{-4}$\\
        \bottomrule
      \end{tabular}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{La pendiente ponderada}
  \begin{codigo}
a_nueva = (m(x * y) - m(x) * m(y)) / (m(x * x) - m(x) ** 2)
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{formula}[Mínimos cuadrados ponderados]
        \begin{equation*}
          a=\frac{\langle xy\rangle-\langle x\rangle\langle y\rangle}
                 {\langle x^{2}\rangle-\langle x\rangle^{2}},
          \qquad
          b=\langle y\rangle-a\langle x\rangle
        \end{equation*}
      \end{formula}
      \small Sale de minimizar
      $\chi^{2}=\sum_i w_i\,(y_i-ax_i-b)^{2}$ respecto de $a$ y $b$. Arriba hay
      una «covarianza» entre $x$ e $y$; abajo, una «varianza» de $x$, ambas
      ponderadas.
    \end{column}
    \begin{column}{0.44\textwidth}
      \small Para la recta por el origen la idea es la misma:
      \begin{codigo}[listing options={style=python, basicstyle=\ttfamily\scriptsize\color{codtexto}}]
a_nueva = (w*x*y).sum() / (w*x*x).sum()
      \end{codigo}
      \begin{destacado}
        \small Sin pesos ($w_i$ iguales) recuperamos las fórmulas básicas de las
        notas: es el mismo ajuste de siempre, solo que cada punto cuenta según
        su fiabilidad.
      \end{destacado}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{¿Cuándo paramos de iterar?}
  \begin{codigo}
if abs(a_nueva - a) <= 1e-12 * max(1.0, abs(a)):
    a = a_nueva
    break
a = a_nueva
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small Si la pendiente nueva apenas difiere de la anterior (una
      billonésima relativa), ha convergido y salimos con \cod{break}. Si no, la
      guardamos y damos otra vuelta. El \cod{range(500)} es solo un tope de
      seguridad.
      \begin{destacado}
        \small El salto de $82$ a $244$\,m/s en la primera vuelta es el efecto de
        los pesos: el punto anómalo deja de arrastrar la recta.
      \end{destacado}
    \end{column}
    \begin{column}{0.44\textwidth}

\begin{frame}[fragile]{Los pesos finales: quién manda en la recta}
  \begin{codigo}
ue = np.sqrt(uy**2 + a**2 * ux**2)
w = 1 / ue**2
  \end{codigo}
  \begin{columns}[T]

      \small Con la pendiente ya convergida recalculo la incertidumbre efectiva

      \begin{destacado}
        \small El punto anómalo prácticamente no cuenta, y las series a

\begin{frame}[fragile]{La ordenada y las incertidumbres de los parámetros}
  \begin{codigo}
b = m(y) - a * m(x)
ua = 1 / math.sqrt(W) / math.sqrt(m(x * x) - m(x) ** 2)
ub = ua * math.sqrt(m(x * x))
nu = n - 2
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{formula}[Notas de incertidumbres, ec.\ (64)]
        \begin{gather*}
          u(a)=\frac{1}{\sqrt{\sum_i w_i}\,\sqrt{\langle x^{2}\rangle-\langle x\rangle^{2}}}\\[2pt]
          u(b)=u(a)\sqrt{\langle x^{2}\rangle}
        \end{gather*}
      \end{formula}
    \end{column}
    \begin{column}{0.42\textwidth}
      \footnotesize
      \begin{itemize}
        \item $b=\langle y\rangle-a\langle x\rangle$: la recta pasa por el
          «centro de gravedad» ponderado. Da $b=-0{,}1014$\,m/s.
        \item $u(a)=7{,}67$\,m/s y $u(b)=0{,}0062$\,m/s: más peso total y más
          separación en $x$ fijan mejor la pendiente.
        \item $\nu=6-2=4$, porque estimamos dos parámetros.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{La bondad del ajuste: $\chi^{2}$}
  \begin{codigo}
chi2 = float((w * (y - a * x - b) ** 2).sum())
factor = max(1.0, math.sqrt(chi2 / nu)) if nu > 0 else 1.0
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{formula}[Chi-cuadrado]
        \begin{equation*}
          \chi^{2}=\sum_i\left(\frac{y_i-ax_i-b}{u_i}\right)^{2}
        \end{equation*}
      \end{formula}
      \small Mide cuántas incertidumbres se separa cada punto de la recta, al
      cuadrado y sumado. Si las barras están bien estimadas,
      $\chi^{2}\approx\nu$.
    \end{column}
    \begin{column}{0.46\textwidth}
      \small En el caso 1, $\chi^{2}=69{,}3$ frente a $\nu=4$: unas 17 veces
      más. Los puntos se dispersan más de lo que dicen sus barras, así que
      $u(a)=7{,}67$\,m/s es demasiado optimista.
      \begin{destacado}
        \small Corregimos con $\sqrt{69{,}3/4}=4{,}16$. El \cod{max(1.0, ...)}
        hace que la incertidumbre solo se infle, nunca se encoja.
      \end{destacado}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{La $t$ de Student y lo que devuelve la función}
  \begin{codigo}
t95 = float(stats.t.ppf(0.975, nu)) if nu > 0 else float("nan")
return dict(a=a, b=b, ua=ua * factor, ub=ub * factor,
            Ua=t95 * ua * factor, Ub=t95 * ub * factor, ...)
  \end{codigo}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \small \cod{stats.t.ppf(0.975, 4)} da el valor que deja un 2{,}5\,\% en
      cada cola: un intervalo central del 95\,\%. Con solo 4 grados de
      libertad, $t_{95,4}=2{,}78$, bastante mayor que el $1{,}96$ de la
      gaussiana, porque con pocos puntos la propia incertidumbre está mal
      conocida.
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering\footnotesize
      \setlength{\tabcolsep}{4pt}
      \renewcommand{\arraystretch}{1.22}
      \begin{tabular}{@{}lll@{}}
        \toprule
        Clave & Contenido & Caso 1\\
        \midrule
        \texttt{a} & pendiente & $244{,}657$\,m/s\\
        \texttt{ua} & $u(a)$ inflada & $7{,}67\times4{,}16=31{,}9$\,m/s\\
        \texttt{Ua} & $t_{95,4}\,u(a)$ & $2{,}78\times31{,}9=88{,}6$\,m/s\\
        \texttt{b}, \texttt{Ub} & ordenada & $-0{,}101\pm0{,}071$\,m/s\\
        \bottomrule
      \end{tabular}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{En resumen}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{formula}[El redondeo]
        \small
        \begin{enumerate}
          \item El logaritmo y \texttt{floor} localizan la primera cifra.
          \item Dividir y truncar nos dice cuál es.
          \item Si es un 1, guardamos dos cifras; si no, una.
          \item Valor e incertidumbre se cortan en la misma posición y se
            escriben con coma decimal.
        \end{enumerate}
      \end{formula}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{formula}[El ajuste ponderado]
        \small
        \begin{enumerate}
          \item Cada punto pesa $1/u_i^{2}$, con la varianza efectiva.
          \item Se itera hasta que la pendiente no cambia.
          \item Se calculan $b$, $u(a)$ y $u(b)$ con medias ponderadas.
          \item $\chi^{2}/\nu$ nos dice si inflar las incertidumbres.
          \item La $t$ de Student da el intervalo al 95\,\%.
        \end{enumerate}
      \end{formula}
    \end{column}
  \end{columns}
\end{frame}
